        function handleModuleUpload(e) {
            e.preventDefault();
            const title = document.getElementById('modulTitle').value;
            const category = document.getElementById('modulCategory').value;
            const fileInput = document.getElementById('modulFile');
            const file = fileInput.files[0];

            if (!file) {
                showModal('error', 'File Belum Dipilih', 'Silakan pilih file dokumen terlebih dahulu.');
                return;
            }

            const fileSizeMB = (file.size / (1024 * 1024)).toFixed(1) + ' MB';
            const fileExt = file.name.split('.').pop().toLowerCase();

            const reader = new FileReader();
            reader.onload = function(event) {
                const newModule = {
                    id: 'mod-' + Date.now(),
                    title: file.name.includes(title) ? file.name : `${title}.${fileExt}`,
                    category: category,
                    size: fileSizeMB,
                    date: new Date().toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' }),
                    type: fileExt,
                    fileData: event.target.result // Base64 Data URL
                };

                appModules.push(newModule);
                localStorage.setItem('simba_modules', JSON.stringify(appModules));
                renderModules();
                closeUploadModal();
                showModal('success', 'Berhasil Unggah', 'Dokumen modul baru berhasil diunggah dan disimpan ke sistem.');
            };
            reader.readAsDataURL(file);
        }

        function viewModule(id) {
            const mod = appModules.find(m => m.id === id);
            if(mod) {
                if(mod.fileData) {
                    const w = window.open('');
                    w.document.write(`<iframe src="${mod.fileData}" frameborder="0" style="border:0; top:0; left:0; bottom:0; right:0; width:100%; height:100%;" allowfullscreen></iframe>`);
                } else {
                    showModal('info', 'Pratinjau Dokumen Resmi', `Menampilkan ringkasan pedoman:\n\n• Judul: ${mod.title}\n• Kategori: ${mod.category}\n• Tanggal: ${mod.date}\n\n(Dokumen standar operasional dan panduan kepatuhan antibiotik RS Tk.II Robert Wolter Mongisidi)`);
                }
            }
        }

        function downloadModule(id) {
            const mod = appModules.find(m => m.id === id);
            if(mod) {
                if(mod.fileData) {
                    const a = document.createElement('a');
                    a.href = mod.fileData;
                    a.download = mod.title;
                    document.body.appendChild(a);
                    a.click();
                    a.remove();
                } else {
                    const contentText = `=== DOKUMEN RESMI SIMBA PPRA ===\nRumah Sakit Tk.II Robert Wolter Mongisidi\n\nJudul: ${mod.title}\nKategori: ${mod.category}\nTanggal Publikasi: ${mod.date}\n\nDokumen ini merupakan panduan resmi yang berlaku di lingkungan rumah sakit.`;
                    const blob = new Blob([contentText], {type: 'text/plain;charset=utf-8'});
                    const url = URL.createObjectURL(blob);
                    const a = document.createElement('a');
                    a.href = url;
                    a.download = mod.title.includes('.') ? mod.title : mod.title + '.txt';
                    document.body.appendChild(a);
                    a.click();
                    a.remove();
                    URL.revokeObjectURL(url);
                    showModal('success', 'Unduh Berhasil', `File "${mod.title}" berhasil diunduh.`);
                }
            }
        }

        function calculateCompliance() {
            let totalY = 0;
            let totalN = 0;
            let answeredCount = 0;
            
            evaluasiSelects.forEach(select => {
                if (select.value) {
                    answeredCount++;
                    if (select.value === 'Y') totalY++;
                    if (select.value === 'N') totalN++;
                }
            });

            const hasilText = document.getElementById('hasilKepatuhanText');
            const percentageText = document.getElementById('persentaseText');
            const circle = document.getElementById('complianceCircle');
            const hiddenStatus = document.getElementById('statusKepatuhanLengkap');
            const hiddenNilai = document.getElementById('nilaiPersentase');

            const totalEvaluated = totalY + totalN;

            if (answeredCount === 4 && totalEvaluated > 0) {
                const percentage = (totalY / totalEvaluated) * 100;
                const roundedPercentage = Math.round(percentage);
                
                percentageText.textContent = `${roundedPercentage}%`;
                hiddenNilai.value = roundedPercentage;

                const circumference = 176;
                const offset = circumference - (percentage / 100) * circumference;
                circle.style.strokeDashoffset = offset;

                if (roundedPercentage === 100) {
                    hasilText.textContent = 'SESUAI';
                    hasilText.className = 'text-2xl sm:text-3xl font-extrabold text-green-600 drop-shadow-sm transition-colors duration-300';
                    circle.classList.remove('text-red-500', 'text-yellow-500');
                    circle.classList.add('text-green-500');
                    hiddenStatus.value = 'SESUAI';
                } else {
                    hasilText.textContent = 'TIDAK SESUAI';
                    hasilText.className = 'text-2xl sm:text-3xl font-extrabold text-red-600 drop-shadow-sm transition-colors duration-300';
                    circle.classList.remove('text-green-500', 'text-yellow-500');
                    circle.classList.add('text-red-500');
                    hiddenStatus.value = 'TIDAK SESUAI';
                }
            } else {
                hasilText.textContent = 'Belum Lengkap';
                hasilText.className = 'text-2xl sm:text-3xl font-extrabold text-gray-400 drop-shadow-sm transition-colors duration-300';
                percentageText.textContent = '0%';
                circle.style.strokeDashoffset = 176;
                circle.classList.remove('text-red-500', 'text-yellow-500', 'text-green-500');
                circle.classList.add('text-gray-300');
                hiddenStatus.value = '';
                hiddenNilai.value = '0';
            }
        }

            document.getElementById('ppraForm').reset();

        function handleFormSubmit(e) {
            e.preventDefault();
            
            if(document.getElementById('statusKepatuhanLengkap').value === '') {
                showModal('error', 'Evaluasi Belum Lengkap', 'Pastikan semua 4 indikator kesesuaian PPK telah dipilih (Y/N).');
                return;
            }

            const abVal = document.getElementById('jenisAntibiotik').value;
            const finalAb = abVal === 'Lainnya' ? document.getElementById('antibiotikLainnya').value : abVal;

            const editingId = document.getElementById('editingId').value;
            const jenisUsiaVal = document.getElementById('jenisUsia').value;
            const bbVal = document.getElementById('beratBadan').value;

            const newData = {
                id: editingId || Date.now().toString(),
                tanggalInput: new Date().toLocaleDateString('id-ID'),
                kodePasien: document.getElementById('kodePasien').value.toUpperCase(),
                tglMasuk: document.getElementById('tglMasuk').value,
                tglKeluar: document.getElementById('tglKeluar').value,
                jenisUsia: jenisUsiaVal,
                beratBadan: bbVal || '-',
                ruangRawat: document.getElementById('ruangRawat').value,
                dxAwal: document.getElementById('dxAwal').value,
                dxAkhir: document.getElementById('dxAkhir').value || '-',
                jenisAntibiotik: finalAb,
                evalJenis: document.getElementById('evalJenis').value,
                evalDosis: document.getElementById('evalDosis').value,
                evalInterval: document.getElementById('evalInterval',).value,
                evalRute: document.getElementById('evalRute').value,
                statusKepatuhan: document.getElementById('statusKepatuhanLengkap').value,
                persentase: document.getElementById('nilaiPersentase').value
            };

            if (editingId) {
                const index = appData.findIndex(item => item.id === editingId);
                if (index !== -1) {
                    appData[index] = newData;
                    localStorage.setItem('simba_data', JSON.stringify(appData));
                    
                    // Trigger sinkronisasi update ke GSheet
                    syncToGoogleSheets(newData, false);

                    showModal('success', 'Berhasil Diperbarui', 'Data evaluasi pasien ' + newData.kodePasien + ' berhasil diperbarui.');
                    resetForm();
                    showView('rekapan');
                    return;
                }
            }

            appData.push(newData);
            localStorage.setItem('simba_data', JSON.stringify(appData));
            
            // Trigger sinkronisasi append ke GSheet
            syncToGoogleSheets(newData, false);

            showModal('success', 'Berhasil Disimpan', 'Data evaluasi pasien ' + newData.kodePasien + ' berhasil ditambahkan.');
            resetForm();
        }

        function editData(id) {
            const item = appData.find(d => d.id === id);
            if (!item) return;

            document.getElementById('editingId').value = item.id;
            document.getElementById('kodePasien').value = item.kodePasien || '';
            document.getElementById('tglMasuk').value = item.tglMasuk || '';
            document.getElementById('tglKeluar').value = item.tglKeluar || '';
            document.getElementById('jenisUsia').value = item.jenisUsia || '';
            document.getElementById('beratBadan').value = item.beratBadan === '-' ? '' : item.beratBadan;

            document.getElementById('ruangRawat').value = item.ruangRawat || '';
            document.getElementById('dxAwal').value = item.dxAwal || '';
            document.getElementById('dxAkhir').value = item.dxAkhir === '-' ? '' : item.dxAkhir;

            const abSelect = document.getElementById('jenisAntibiotik');
            const options = Array.from(abSelect.options).map(o => o.value);
            if (options.includes(item.jenisAntibiotik)) {
                abSelect.value = item.jenisAntibiotik;
                handleAntibiotikLainnya();
            } else {
                abSelect.value = 'Lainnya';
                handleAntibiotikLainnya();
                document.getElementById('antibiotikLainnya').value = item.jenisAntibiotik;
            }

            document.getElementById('evalJenis').value = item.evalJenis || '';
            document.getElementById('evalDosis').value = item.evalDosis || '';
            document.getElementById('evalInterval').value = item.evalInterval || '';
            document.getElementById('evalRute').value = item.evalRute || '';

            calculateCompliance();

            document.getElementById('formSubmitBtnText').textContent = 'Perbarui Data Evaluasi';
            document.getElementById('formTitle').innerHTML = '<i class="fas fa-edit text-blue-600 mr-2.5"></i> Edit Data Evaluasi PPRA';

            showView('inputForm');
        }

        function renderTable() {
            const tbody = document.getElementById('tableBody');
            const emptyState = document.getElementById('emptyState');
            const searchVal = document.getElementById('searchInput').value.toLowerCase();
            
            tbody.innerHTML = '';
            
            let filteredData = appData.filter(item => 
                item.kodePasien.toLowerCase().includes(searchVal) || 
                item.jenisAntibiotik.toLowerCase().includes(searchVal)
            );

            filteredData.sort((a, b) => b.id - a.id);

            let statSesuai = 0;
            let statTidak = 0;

            if (filteredData.length === 0) {
                emptyState.classList.remove('hidden');
                emptyState.classList.add('flex');
            } else {
                emptyState.classList.add('hidden');
                emptyState.classList.remove('flex');

                filteredData.forEach(item => {
                    if(item.statusKepatuhan === 'SESUAI') statSesuai++; else statTidak++;

                    const tr = document.createElement('tr');
                    tr.className = 'hover:bg-gray-50 transition-colors';
                    
                    const statusClass = item.statusKepatuhan === 'SESUAI' 
                        ? 'bg-green-100 text-green-800' 
                        : 'bg-red-100 text-red-800';
                        
                    const evalString = `${item.evalJenis}/${item.evalDosis}/${item.evalInterval}/${item.evalRute}`;
                    const bbDisplay = item.beratBadan && item.beratBadan !== '-' ? item.beratBadan + ' kg' : '-';

                    tr.innerHTML = `
                        <td class="px-4 py-3 whitespace-nowrap text-xs font-medium">
                            <div class="flex space-x-1.5">
                                <button onclick="editData('${item.id}')" class="text-blue-600 hover:text-blue-900 bg-blue-50 p-1.5 rounded-lg border border-blue-100 shadow-sm" title="Edit"><i class="fas fa-edit"></i></button>
                                <button onclick="deleteDataPrompt('${item.id}')" class="text-red-600 hover:text-red-900 bg-red-50 p-1.5 rounded-lg border border-red-100 shadow-sm" title="Hapus"><i class="fas fa-trash-alt"></i></button>
                            </div>

                        <td class="px-4 py-3 whitespace-nowrap text-xs font-bold text-gray-800">${item.kodePasien}</td>

                        <td class="px-4 py-3 whitespace-nowrap text-xs font-semibold text-gray-800">${item.ruangRawat}</td>

                            <span class="px-2.5 py-1 inline-flex text-[11px] leading-tight font-bold rounded-full ${statusClass}">
                                ${item.statusKepatuhan}
                            </span>
                        </td>
                    `;
                    tbody.appendChild(tr);
                });
            }

            document.getElementById('filterSesuai').textContent = statSesuai;
            document.getElementById('filterTidakSesuai').textContent = statTidak;
        }

            showModal('confirm', 'Hapus Seluruh Data', 'PERINGATAN: Tindakan ini akan menghapus semua rekapan data pasien secara permanen. Lanjutkan?', () => {
                appData = [];
                localStorage.removeItem('simba_data');
                renderTable();
                updateDashboardStats();
                showModal('success', 'Database Direset', 'Semua data rekapan PPRA telah dihapus.');
            });
        }

        function backupLocalData() {
            if (appData.length === 0) {
                showModal('info', 'Data Kosong', 'Belum ada data untuk dicadangkan.');
                return;
            }
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(appData, null, 2));
            const dlAnchor = document.createElement('a');
            dlAnchor.setAttribute("href", dataStr);
            dlAnchor.setAttribute("download", `SIMBA_PPRA_Backup_${new Date().toISOString().split('T')[0]}.json`);
            document.body.appendChild(dlAnchor);
            dlAnchor.click();
            dlAnchor.remove();
            showModal('success', 'Cadangan Berhasil', 'File cadangan JSON berhasil diunduh.');
        }

        function updateDashboardStats() {
            if(!appData || appData.length === 0) {
                document.getElementById('stat-total').textContent = '0';
                document.getElementById('stat-compliance').textContent = '0%';
                document.getElementById('stat-top-ab').textContent = '-';
                document.getElementById('stat-top-room').textContent = '-';
                return;
            }

            document.getElementById('stat-total').textContent = appData.length;

            let totalP = 0;
            const abCounts = {};
            const roomCounts = {};

            appData.forEach(item => {
                totalP += parseInt(item.persentase || 0);
                abCounts[item.jenisAntibiotik] = (abCounts[item.jenisAntibiotik] || 0) + 1;
                roomCounts[item.ruangRawat] = (roomCounts[item.ruangRawat] || 0) + 1;
            });

            const avgP = Math.round(totalP / appData.length);
            const compEl = document.getElementById('stat-compliance');
            compEl.textContent = `${avgP}%`;
            compEl.className = `text-xl sm:text-2xl font-bold ${avgP >= 80 ? 'text-green-600' : 'text-yellow-600'}`;

            let maxAb = '-', maxAbCount = 0;
            for (const [key, value] of Object.entries(abCounts)) {
                if(value > maxAbCount) { maxAbCount = value; maxAb = key; }
            }
            document.getElementById('stat-top-ab').textContent = maxAb;

            let maxRoom = '-', maxRoomCount = 0;
            for (const [key, value] of Object.entries(roomCounts)) {
                if(value > maxRoomCount) { maxRoomCount = value; maxRoom = key; }
            }
            document.getElementById('stat-top-room').textContent = maxRoom;
        }

        function exportToPDF() {
            if (appData.length === 0) {
                showModal('info', 'Data Kosong', 'Tidak ada data untuk diekspor.');
                return;
            }

            const { jsPDF } = window.jspdf;
            const doc = new jsPDF('l', 'mm', 'a4');
            
            doc.setFontSize(14);
            doc.setFont('helvetica', 'bold');
            doc.text("LAPORAN EVALUASI KEPATUHAN ANTIBIOTIK (PPRA)", 14, 15);
            doc.setFontSize(10);
            doc.setFont('helvetica', 'normal');
            doc.text(`RS TK.II Robert Wolter Mongisidi | Tanggal Cetak: ${new Date().toLocaleDateString('id-ID')}`, 14, 22);

            const head = [['Code Pasien', 'Usia/BB', 'Ruang Rawat', 'Antibiotik', 'Sesuai PPK', 'Dosis', 'Interval', 'Rute', 'Status', '%']];
            const body = appData.map(item => [
                item.kodePasien,
                `${item.jenisUsia}\n${item.beratBadan === '-' ? '' : item.beratBadan+'kg'}`,
                item.ruangRawat,
                item.jenisAntibiotik,
                item.evalJenis,
                item.evalDosis,
                item.evalInterval,
                item.evalRute,
                item.statusKepatuhan,
                `${item.persentase}%`
            ]);

            doc.autoTable({
                head: head,
                body: body,
                startY: 30,
                theme: 'grid',
                styles: { fontSize: 8, cellPadding: 2 },
                headStyles: { fillColor: [22, 163, 74] },
                didParseCell: function(data) {
                    if (data.section === 'body' && data.column.index === 8) {
                        if (data.cell.raw === 'SESUAI') {
                            data.cell.styles.textColor = [22, 163, 74];
                            data.cell.styles.fontStyle = 'bold';
                        } else {
                            data.cell.styles.textColor = [220, 38, 38];
                            data.cell.styles.fontStyle = 'bold';
                        }
                    }
                }
            });

            const dateStr = new Date().toISOString().split('T')[0];
            doc.save(`Laporan_SIMBA_PPRA_${dateStr}.pdf`);
        }

        // Fungsi utama mengirim data ke GSheet secara asinkron
        function syncToGoogleSheets(data, isBulk = false) {
            const gasUrl = localStorage.getItem('simba_gas_url');
            if (!gasUrl) return;

            const originalText = isBulk ? document.getElementById('syncBtnText').innerHTML : '';
            if (isBulk) {
                document.getElementById('syncBtnText').innerHTML = '<i class="fas fa-spinner fa-spin mr-1.5"></i> Menyinkronkan ke Cloud...';
                document.getElementById('syncBtn').disabled = true;
            }

            // Membungkus data dengan flag action (single atau bulk upload)
            const payload = isBulk ? { action: 'bulk', data: data } : { action: 'single', data: data };

            // Menggunakan method 'no-cors' karena Apps Script sering membutuhkan ini untuk cross-domain form
            fetch(gasUrl, {
                method: 'POST',
                mode: 'no-cors',
                headers: { 'Content-Type': 'text/plain' }, // Pakai text/plain untuk bypass CORS preflight HTTP
                body: JSON.stringify(payload)
            }).then(() => {
                if (isBulk) {
                    showModal('success', 'Sinkronisasi Selesai', 'Seluruh rekapan data lokal berhasil dikirim dan diamankan ke Google Spreadsheet.');
                    document.getElementById('syncBtnText').innerHTML = originalText;
                    document.getElementById('syncBtn').disabled = false;
                }
            }).catch(err => {
                console.error('Error syncing:', err);
                if (isBulk) {
                    showModal('error', 'Koneksi Gagal', 'Terjadi kesalahan saat menghubungi server Google. Pastikan URL benar dan Anda terhubung ke internet.');
                    document.getElementById('syncBtnText').innerHTML = originalText;
                    document.getElementById('syncBtn').disabled = false;
                }
            });
        }

        // Fungsi untuk trigger tombol Sinkronisasi Semua Data
        function syncAllData() {
            if(appData.length === 0) {
                showModal('info', 'Data Kosong', 'Tidak ada rekapan data lokal untuk disinkronkan.');
                return;
            }
            syncToGoogleSheets(appData, true);
        }
